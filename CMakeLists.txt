    "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>" "$<INSTALL_INTERFACE:include>")

    target_compile_options(dittobind PUBLIC /utf-8)

include(GNUInstallDirs)

    ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})

    "import importlib.metadata as m; d=m.distribution('nanobind'); print(d.locate_file(next(p for p in d.files if str(p).endswith('licenses/LICENSE'))))"
    OUTPUT_VARIABLE nanobind_license OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
file(TO_CMAKE_PATH "${nanobind_license}" nanobind_license)
install(FILES "${nanobind_license}" DESTINATION ${CMAKE_INSTALL_DATADIR}/dittobind/licenses RENAME nanobind.txt)
install(FILES third_party/robin-map-LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/dittobind/licenses)
configure_package_config_file(cmake/dittobindConfig.cmake.in
    "${CMAKE_CURRENT_BINARY_DIR}/dittobindConfig.cmake"
    INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/dittobind)
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/dittobindConfigVersion.cmake"
    VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(EXPORT dittobindTargets NAMESPACE dittobind:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/dittobind)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/dittobindConfig.cmake"
    "${CMAKE_CURRENT_BINARY_DIR}/dittobindConfigVersion.cmake"
    DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/dittobind)

function(dittobind_configure_runner target)
    set_target_properties(${target} PROPERTIES VS_GLOBAL_VcpkgEnabled false)
    target_compile_definitions(${target} PRIVATE DITTOBIND_PYTHON_EXECUTABLE="${Python_EXECUTABLE}")
    target_compile_definitions(${target} PRIVATE DITTOBIND_PYTHON_HOME="${dittobind_python_home}")
    # The executable needs the matching CPython DLL even outside an activated venv.
    if(WIN32)
        add_custom_command(TARGET ${target} POST_BUILD
            COMMAND ${CMAKE_COMMAND} -E copy_if_different
                "$<TARGET_FILE:Python::Python>" "$<TARGET_FILE_DIR:${target}>")
    endif()
endfunction()
